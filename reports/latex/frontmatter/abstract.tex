\chapter*{Abstract}
\addcontentsline{toc}{chapter}{Abstract}
\fillpara{[R1] Tối đa 200 từ, một đoạn, theo thứ tự: bối cảnh (P300, oddball); dữ liệu (Nencki-Symfonia, số subject và epoch dùng được); phương pháp (tiền xử lý, 2 mô hình classical, 2 mô hình DL, within và cross-subject, cửa sổ full và early); kết quả chính (balanced accuracy và AUC tốt nhất, classical so với DL, mô hình dựa vào đặc trưng nào); một câu kết luận.}

\noindent\textbf{Keywords:} \fillin{4 đến 6 từ khoá, ví dụ: P300, visual oddball, single-trial classification, EEGNet, xDAWN, explainability}
