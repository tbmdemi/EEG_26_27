\chapter{Data}
\label{ch:data}

\section{Dataset}
\fillpara{[R2] Nencki-Symfonia \citep{dzianok2022}, OpenNeuro ds004621: số người tham gia, độ tuổi, thiết bị ghi (số kênh, tần số lấy mẫu, hệ điện cực, tham chiếu khi ghi). Nguồn: \src{docs/02_dataset.md}.}

\section{Task design}
\fillpara{[R2] Visual oddball: 3 loại kích thích (standard S5, target S6, distractor S7), tỉ lệ từng loại, thời gian trình bày, khoảng cách giữa các kích thích, tay bấm nút (tay phải), các lần nghỉ (S 10). Có thể thêm hình minh hoạ trình tự trial.}

\section{Behavioural scoring}
\fillpara{[R2] Quy tắc giữ trial đúng: target có phản hồi trong 100 đến 1400 ms sau kích thích; standard và distractor không có phản hồi trong khoảng đó. RT lấy từ phản hồi đầu tiên. Nguồn: \src{reports/report_template.md} mục 3.1.}

\section{Participants and exclusions}
\fillpara{[R2] Loại trước sub-02, sub-14, sub-17 (hiểu sai nhiệm vụ, theo bài báo); loại theo tiêu chí pipeline (dưới 40 target đúng, hoặc giữ dưới 50\% target sau khi loại epoch); các subject chỉ gắn cờ. Bảng chi tiết từng subject đặt ở Phụ lục \ref{app:subjects}.}
